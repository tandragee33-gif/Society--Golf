function calculateTeamResults() { // Convert the DD/MM/YYYY input to YYYY-MM-DD so it matches the database const rawDate = document.getElementById('tmDate').value.trim(); const matchDate = parseDateInput(rawDate); if (competitionTeams.length === 0) { alert("You haven't added any teams yet!"); return; } const results = competitionTeams.map(team => { let teamTotal = 0; for (let i = 0; i < 18; i++) { let holeScores = team.members.map(m => { const r = rounds.find(rd => rd.date === matchDate && String(rd.playerId) === String(m.id) ); return (r && r.pointsArray) ? r.pointsArray[i] : 0; }); holeScores.sort((a, b) => b - a); // Holes 9 (index 8) and 18 (index 17) if (i === 8 || i === 17) { teamTotal += holeScores.reduce((a, b) => a + b, 0); } else { teamTotal += (holeScores[0] || 0) + (holeScores[1] || 0); } } return { name: `Team ${team.id}`, score: teamTotal, members: team.members }; }); // Output the leaderboard const output = document.getElementById('team-results-output'); output.innerHTML = `

Leaderboard for ${rawDate}

` + results.map(r => `
${r.name}
${r.members.map(m=>m.name).join(', ')} ${r.score} pts
`).join(''); }
